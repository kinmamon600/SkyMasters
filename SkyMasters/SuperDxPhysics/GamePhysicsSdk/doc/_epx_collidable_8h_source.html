23 #ifndef EPX_COLLIDABLE_H
24 #define EPX_COLLIDABLE_H

26 #include "../EpxBase.h"

29 #define EPX_NUM_SHAPES 5

31 namespace EasyPhysics {
32 
34 struct EpxCollidable {
35  EpxUInt8 m_numShapes;
36  EpxShape m_shapes[EPX_NUM_SHAPES];
37  EpxVector3 m_center;
38  EpxVector3 m_half;
39 
41  void reset()
42  {
43  m_numShapes = 0;
44  m_center = EpxVector3(0.0f);
45  m_half = EpxVector3(0.0f);
46  }
47 
51  void addShape(const EpxShape &shape)
52  {
53  if(m_numShapes < EPX_NUM_SHAPES) {
54  m_shapes[m_numShapes++] = shape;
55  }
56  }
57 
60  void finish()
61  {
62  // AABBを計算する
63  EpxVector3 aabbMax(-EPX_FLT_MAX),aabbMin(EPX_FLT_MAX);
64  for(EpxUInt32 i=0;i<m_numShapes;i++) {
65  const EpxConvexMesh &mesh = m_shapes[i].m_geometry;
66 
67  for(EpxUInt32 v=0;v<mesh.m_numVertices;v++) {
68  aabbMax = maxPerElem(aabbMax,m_shapes[i].m_offsetPosition + rotate(m_shapes[i].m_offsetOrientation,mesh.m_vertices[v]));
69  aabbMin = minPerElem(aabbMin,m_shapes[i].m_offsetPosition + rotate(m_shapes[i].m_offsetOrientation,mesh.m_vertices[v]));
70  }
71  }
72  m_center = (aabbMax + aabbMin) * 0.5f;
73  m_half = (aabbMax - aabbMin) * 0.5f;
74  }
75 };

77 } // namespace EasyPhysics
